# 20260511.md — UpUp Agent SDK 全面改造计划

> 创建日期: 2026-05-11 | 版本: 3.0 | 目标: 对标 Claude Code Agent SDK + Codex SDK，构建对外 UP SDK

---

## 执行摘要

本计划旨在将 UpUp 改造为类似 Claude Code 的 Agent SDK 架构，参考 Claude Agent SDK 和 Codex (Elixir) SDK 的最佳实践，构建可对外发布的 UP SDK。

### 核心目标

| 目标 | 描述 | 优先级 |
|------|------|--------|
| `@upup/agent-core` | 独立的 Agent 运行时核心包 | P0 |
| `@upup/sdk` | 对外集成 SDK (npm 发布) | P0 |
| Agent 定义文件 | YAML frontmatter 格式的 Agent 定义 | P0 |
| 工具标准化 | 标准 Tool 接口 + 工具限制 | P0 |
| Hooks 系统 | PreToolUse/PostToolUse/Stop 事件 | P1 |
| 子代理系统 | 动态 Agent 生成 + Task 工具 | P1 |
| MCP 集成 | 完整 MCP 协议支持 | P2 |

### 1.2 现有核心能力

**Agent 核心 (src/agent/):**
- `agent.ts` (43KB) - Agent 主循环，支持流式输出、并发执行、循环检测、压缩
- `tool-executor.ts` - 工具执行器，支持并发执行
- `subagent-runner.ts` (17KB) - 子代理运行器，支持后台任务
- `compact.ts` - 上下文压缩
- `loop-recovery.ts` - 循环检测与恢复
- `session-persistence.ts` - 会话持久化
- `registry.ts` - 工具注册表
- `scratchpad.ts` - 草稿板

**工具系统 (src/tools/):**
- 64+ 工具，涵盖文件系统、bash、金融、浏览器、缓存等
- 基于 LangChain 的 StructuredToolInterface
- 工具描述生成、结果预算管理

**记忆系统 (src/memory/):**
- 观察缓冲、记忆提取、上下文压缩
- 基于 DuckDB 和 MemVid 的混合存储

**MCP 集成 (packages/mcp/):**
- 使用官方 @modelcontextprotocol/sdk
- 支持 Stdio 和 SSE 传输

**插件系统:**
- `@upup/plugin-sdk` - 工具注册、钩子、CLI 命令、MCP 通道
- `@upup/plugins` - manifest 验证、插件类型

## 4. 核心接口设计

### 4.1 标准工具接口

```typescript
// packages/agent-core/src/tool.ts
import { z } from 'zod';

/**
 * 标准工具接口 - 对标 Claude Code Tool 接口
 */
export interface Tool<Name extends string = string> {
  /** 工具唯一名称 */
  name: Name;

  /** 工具描述，用于 LLM 理解工具用途 */
  description: string;

  /** 输入 Schema (Zod) */
  inputSchema: z.ZodType;

  /** 执行函数 */
  execute(input: unknown, context?: ToolContext): Promise<ToolResult>;
}

/** 工具执行上下文 */
export interface ToolContext {
  sessionId: string;
  toolCallId: string;
  userId?: string;
  cwd: string;
  signal?: AbortSignal;
}

/** 工具执行结果 */
export interface ToolResult {
  success: boolean;
  output?: string;
  error?: string;
  metadata?: Record<string, unknown>;
}

/** 工具分类 */
export type ToolCategory =
  | 'filesystem'   // 文件操作
  | 'bash'         // 命令行
  | 'code'         // 代码生成
  | 'search'       // 搜索
  | 'web'          // 网络请求
  | 'database'     // 数据库
  | 'mcp'          // MCP 工具
  | 'custom';      // 自定义

/** 工具元数据 */
export interface ToolMetadata {
  name: string;
  category: ToolCategory;
  tags: string[];
  author?: string;
  version?: string;
  isBuiltin?: boolean;
  description: string;
}

/** LangChain 工具转换为标准工具 */
export function fromLangChainTool(
  tool: StructuredToolInterface
): Tool {
  return {
    name: tool.name as Tool['name'],
    description: tool.description,
    inputSchema: tool.schema as z.ZodType,
    execute: async (input) => {
      try {
        const result = await tool.invoke(input);
        return { success: true, output: String(result) };
      } catch (error) {
        return {
          success: false,
          error: error instanceof Error ? error.message : String(error)
        };
      }
    }
  };
}

/** 标准工具转换为 LangChain 工具 */
export function toLangChainTool(tool: Tool): StructuredToolInterface {
  return new DynamicStructuredTool({
    name: tool.name,
    description: tool.description,
    schema: tool.inputSchema,
    execute: async (input) => {
      const result = await tool.execute(input);
      if (!result.success) {
        throw new Error(result.error);
      }
      return result.output;
    }
  });
}
```

```typescript
const FrontmatterSchema = z.object({
  name: z.string().optional(),
  description: z.string().optional(),
  model: z.string().optional(),
  color: z.enum(['blue', 'red', 'green', 'yellow', 'purple', 'orange']).optional(),

  author: z.string().optional(),
  version: z.string().optional(),

export interface Frontmatter {
  name?: string;
  description?: string;
  model?: string;
  color?: 'blue' | 'red' | 'green' | 'yellow' | 'purple' | 'orange';
  tools?: string[];
  author?: string;
  version?: string;
}

export function parseFrontmatter(content: string): { frontmatter: Frontmatter; body: string } {
  const match = content.match(/^---\n([\s\S]*?)\n---\n([\s\S]*)$/);
  if (!match) {
    return { frontmatter: {}, body: content };
  }

  const yamlContent = match[1];
  const body = match[2];

  // 简单 YAML 解析 (生产环境使用 js-yaml)
  const frontmatter = parseYaml(yamlContent);

  return { frontmatter, body };
}

function parseYaml(yaml: string): Frontmatter {
  const result: Record<string, unknown> = {};

  for (const line of yaml.split('\n')) {
    const colonIndex = line.indexOf(':');
    if (colonIndex === -1) continue;

    const key = line.slice(0, colonIndex).trim();
    let value: unknown = line.slice(colonIndex + 1).trim();

    // 解析数组
    if (value.startsWith('[') && value.endsWith(']')) {
      value = value.slice(1, -1).split(',').map(s => s.trim().replace(/^["']|["']$/g, ''));
    }

    result[key] = value;
  }

  return FrontmatterSchema.parse(result);
}

export function parseAgentFile(content: string, filePath: string): AgentDefinition {
  const { frontmatter, body } = parseFrontmatter(content);

  return {
    path: filePath,
    metadata: {
      id: frontmatter.name ?? path.basename(filePath, '.md'),
      name: frontmatter.name ?? path.basename(filePath, '.md'),
      description: frontmatter.description ?? '',
      model: frontmatter.model,
      color: frontmatter.color,
      tools: frontmatter.tools ?? ['*'],
      author: frontmatter.author,
      version: frontmatter.version,
      createdAt: Date.now(),
      updatedAt: Date.now(),
    },
    systemPrompt: body.trim(),
  };
}
```

```typescript
// packages/agent-core/src/agent.ts
import type { AgentConfig, RunOptions, RunResult, AgentEvent, ToolCall, TokenUsage } from './types.js';
import type { Tool } from './tool.js';
import type { HookContext, HookResult } from './hooks/types.js';
import { getToolRegistry } from './registry.js';
import { HookManager } from './hooks/hook-manager.js';
import { callLlmWithMessages, streamLlmWithMessages } from '@upup/llm';
import { extractToolCalls, extractTextContent } from './utils/ai-message.js';

const DEFAULT_MAX_ITERATIONS = 50;

export class Agent {
  readonly config: AgentConfig;
  private tools: Tool[];
  private stopped = false;
  private hookManager: HookManager;

  constructor(config: AgentConfig, options?: { tools?: Tool[]; hooks?: HookManager }) {
    this.config = config;
    this.tools = options?.tools ?? getToolRegistry().list();
    this.hookManager = options?.hooks ?? new HookManager(config.hooks);
  }

  static async create(config: AgentConfig): Promise<Agent> {
    const registry = getToolRegistry();
    let tools = registry.list();

    // 应用工具过滤
    if (config.tools && config.tools[0] !== '*') {
      const allowed = new Set(config.tools);
      tools = tools.filter(t => allowed.has(t.name));
    }

    return new Agent(config, { tools });
  }

  static fromFile(filePath: string): Promise<Agent> {
    // 读取文件并解析
    return Promise.resolve(new Agent({}));
  }

  async *stream(input: string, options?: RunOptions): AsyncGenerator<AgentEvent> {
    this.stopped = false;
    let iteration = 0;
    const maxIterations = this.config.maxIterations ?? DEFAULT_MAX_ITERATIONS;
    const messages: Message[] = [];

    // 添加系统消息
    if (this.config.systemPrompt) {
      messages.push({ role: 'system', content: this.config.systemPrompt });
    }

    // 添加用户消息
    messages.push({ role: 'user', content: input });

    while (iteration < maxIterations && !this.stopped) {
      iteration++;

      // 发送消息到 LLM
      const response = yield* this.callModel(messages);

      // 检查是否有文本回复
      if (response.text) {
        yield { type: 'text', content: response.text };
      }

      // 检查是否有工具调用
      if (!response.toolCalls || response.toolCalls.length === 0) {
        return {
          answer: response.text ?? '',
          toolCalls: [],
          iterations: iteration,
          totalTime: 0
        };
      }

      // 执行工具
      for (const toolCall of response.toolCalls) {
        // PreToolUse Hook
        const hookResult = await this.hookManager.execute({
          event: 'PreToolUse',
          toolName: toolCall.name,
          args: toolCall.args,
        });

        if (hookResult.blocked || hookResult.permissionDecision === 'deny') {
          yield {
            type: 'tool_denied',
            tool: toolCall.name,
            args: toolCall.args,
            reason: hookResult.systemMessage,
          };
          continue;
        }

        yield { type: 'tool_start', tool: toolCall.name, args: hookResult.updatedArgs ?? toolCall.args };

        const tool = getToolRegistry().get(toolCall.name);
        if (!tool) {
          yield { type: 'tool_error', tool: toolCall.name, error: 'Tool not found' };
          continue;
        }

        try {
          const startTime = Date.now();
          const result = await tool.execute(hookResult.updatedArgs ?? toolCall.args);
          const duration = Date.now() - startTime;

          yield {
            type: 'tool_end',
            tool: toolCall.name,
            result: result.output ?? result.error ?? '',
            duration,
          };

          // 添加工具结果到消息
          messages.push({
            role: 'tool',
            content: result.output ?? result.error ?? '',
          });

          // PostToolUse Hook
          await this.hookManager.execute({
            event: 'PostToolUse',
            toolName: toolCall.name,
            args: toolCall.args,
          });
        } catch (error) {
          yield {
            type: 'tool_error',
            tool: toolCall.name,
            error: error instanceof Error ? error.message : String(error),
          };
        }
      }
    }

    return { answer: '', toolCalls: [], iterations: iteration, totalTime: 0 };
  }

  async run(input: string, options?: RunOptions): Promise<RunResult> {
    const events: AgentEvent[] = [];

    for await (const event of this.stream(input, options)) {
      events.push(event);
    }

    // 从事件中提取结果
    const lastEvent = events[events.length - 1];
    if (lastEvent && lastEvent.type === 'done') {
      return lastEvent as unknown as RunResult;
    }

    return {
      answer: '',
      toolCalls: [],
      iterations: 0,
      totalTime: 0,
    };
  }

  stop(): void {
    this.stopped = true;
  }

  private async *callModel(
    messages: Message[]
  ): AsyncGenerator<AgentEvent, { text?: string; toolCalls?: ToolCall[] }> {
    // 调用 LLM
    const toolDefs = this.tools.map(t => ({
      name: t.name,
      description: t.description,
      inputSchema: t.inputSchema,
    }));

    try {
      const response = await streamLlmWithMessages(messages, {
        model: this.config.model,
        tools: toolDefs,
      });

      const text = extractTextContent(response);
      const toolCalls = extractToolCalls(response);

      return { text, toolCalls };
    } catch (error) {
      return { text: error instanceof Error ? error.message : String(error) };
    }
  }
}
```

#### 4.1 Hook 管理器实现

```typescript
// packages/agent-core/src/hooks/hook-manager.ts
import type { HookConfig, HookResult, HookContext, HookEvent } from './types.js';

export class HookManager {
  private hooks: Map<HookEvent, HookConfig[]> = new Map();

  constructor(configs?: HookConfig[]) {
    if (configs) {
      this.registerAll(configs);
    }
  }

  /** 注册 Hook */
  register(config: HookConfig): void {
    const existing = this.hooks.get(config.event) ?? [];
    existing.push(config);
    this.hooks.set(config.event, existing);
  }

  /** 批量注册 */
  registerAll(configs: HookConfig[]): void {
    for (const config of configs) {
      this.register(config);
    }
  }

  /** 获取所有配置 */
  getConfigs(): HookConfig[] {
    const configs: HookConfig[] = [];
    for (const eventConfigs of this.hooks.values()) {
      configs.push(...eventConfigs);
    }
    return configs;
  }

  /** 执行 Hook */
  async execute(context: HookContext): Promise<HookResult> {
    const configs = this.hooks.get(context.event) ?? [];
    let result: HookResult = { permissionDecision: 'allow' };

    for (const config of configs) {
      // 检查 matcher
      if (!this.matches(config.matcher, context.toolName ?? '')) {
        continue;
      }

      // 执行每个 hook
      for (const hook of config.hooks) {
        const hookResult = await this.executeHook(hook, context);

        // 更新结果
        if (hookResult.permissionDecision === 'deny') {
          result = hookResult;
        }

        // 如果阻止执行
        if (hookResult.blocked || hook.blocking) {
          return hookResult;
        }
      }
    }

    return result;
  }

  private matches(matcher: string, toolName: string): boolean {
    if (matcher === '*' || matcher === '.*') return true;
    if (matcher.startsWith('!')) {
      return !new RegExp(matcher.slice(1)).test(toolName);
    }
    return new RegExp(matcher).test(toolName);
  }

  private async executeHook(
    hook: { type: 'prompt' | 'command'; prompt?: string; command?: string; timeout?: number },
    context: HookContext
  ): Promise<HookResult> {
    if (hook.type === 'command' && hook.command) {
      return this.executeCommand(hook.command, context, hook.timeout);
    }

    // prompt 类型需要 LLM 判断，这里简化处理
    return { permissionDecision: 'allow' };
  }

  private async executeCommand(
    command: string,
    context: HookContext,
    timeout = 30
  ): Promise<HookResult> {
    try {
      // 模板替换
      const filledCommand = command
        .replace(/\{\{toolName\}\}/g, context.toolName ?? '')
        .replace(/\{\{args\}\}/g, JSON.stringify(context.args ?? {}));

      // 执行命令
      const { exec } = await import('child_process');
      const result = await new Promise<string>((resolve, reject) => {
        exec(filledCommand, { timeout: timeout * 1000 }, (error, stdout, stderr) => {
          if (error) reject(error);
          else resolve(stdout || stderr);
        });
      });

      // 解析结果
      try {
        const parsed = JSON.parse(result);
        return {
          permissionDecision: parsed.decision ?? 'allow',
          data: parsed,
        };
      } catch {
        return { permissionDecision: 'allow' };
      }
    } catch (error) {
      return {
        permissionDecision: 'allow',
        error: error instanceof Error ? error.message : String(error),
      };
    }
  }

  /** 清除所有 Hook */
  clear(): void {
    this.hooks.clear();
  }

  /** 移除特定 Hook */
  remove(event: HookEvent, matcher: string): void {
    const configs = this.hooks.get(event) ?? [];
    this.hooks.set(event, configs.filter(c => c.matcher !== matcher));
  }
}
```

```typescript
    try {
      const files = await readdir(dirPath);
      for (const file of files) {
        if (file.endsWith('.md')) {
          const filePath = join(dirPath, file);
          const content = await readFile(filePath, 'utf-8');
          const definition = parseAgentFile(content, filePath);
          definitions.push(definition);
        }
      }
    } catch {
      // 目录不存在
    }

    return definitions;
  }

    const results: AgentDefinition[] = [];

    for (const dir of dirs) {
      const agents = await this.scanAgents(dir);
      results.push(...agents);
    }
```

```typescript
// packages/agent-core/src/task.ts
import type { Agent } from './agent.js';
import type { AgentEvent, RunResult, TaskOptions, TaskResult } from './types.js';
import { getAgentFactory } from './factory.js';

export class TaskRunner {
  private tasks: Map<string, TaskResult> = new Map();
  private runningAgents: Map<string, Agent> = new Map();

  constructor() {}

  /** 运行任务 */
  async run(options: TaskOptions): Promise<string> {
    const taskId = this.generateTaskId();

    // 创建或获取 Agent
    let agent: Agent;
    if (options.subagent_type === 'general-purpose') {
      agent = new Agent({
        systemPrompt: options.prompt,
        tools: options.tools,
      });
    } else {
      const factory = getAgentFactory();
      const builtin = factory.createBuiltin(options.subagent_type);
      if (!builtin) {
        throw new Error(`Unknown subagent type: ${options.subagent_type}`);
      }
      agent = builtin;
    }

    // 保存任务
    this.tasks.set(taskId, {
      id: taskId,
      status: 'running',
      createdAt: Date.now(),
    });
    this.runningAgents.set(taskId, agent);

    // 执行
    if (options.background) {
      this.runInBackground(taskId, agent, options.prompt, options.timeout);
    } else {
      await this.runSynchronous(taskId, agent, options.prompt, options.timeout);
    }

    return taskId;
  }

  private async runSynchronous(
    taskId: string,
    agent: Agent,
    prompt: string,
    timeout?: number
  ): Promise<void> {
    const startTime = Date.now();

    try {
      const result = await agent.run(prompt, { signal: undefined });

      this.tasks.set(taskId, {
        id: taskId,
        status: 'completed',
        output: result.answer,
        duration: Date.now() - startTime,
        createdAt: this.tasks.get(taskId)?.createdAt ?? Date.now(),
        completedAt: Date.now(),
      });
    } catch (error) {
      this.tasks.set(taskId, {
        id: taskId,
        status: 'failed',
        error: error instanceof Error ? error.message : String(error),
        duration: Date.now() - startTime,
        createdAt: this.tasks.get(taskId)?.createdAt ?? Date.now(),
        completedAt: Date.now(),
      });
    } finally {
      this.runningAgents.delete(taskId);
    }
  }

  private async runInBackground(
    taskId: string,
    agent: Agent,
    prompt: string,
    timeout?: number
  ): Promise<void> {
    // 后台运行，不等待完成
    this.runSynchronous(taskId, agent, prompt, timeout).catch(() => {});
  }

  /** 获取任务状态 */
  getTask(taskId: string): TaskResult | undefined {
    return this.tasks.get(taskId);
  }

  /** 取消任务 */
  cancel(taskId: string): boolean {
    const agent = this.runningAgents.get(taskId);
    if (agent) {
      agent.stop();
      this.runningAgents.delete(taskId);
      this.tasks.set(taskId, {
        ...this.tasks.get(taskId)!,
        status: 'cancelled',
        completedAt: Date.now(),
      });
      return true;
    }
    return false;
  }

  /** 列出所有任务 */
  listTasks(filter?: { status?: TaskResult['status'] }): TaskResult[] {
    let tasks = Array.from(this.tasks.values());
    if (filter?.status) {
      tasks = tasks.filter(t => t.status === filter.status);
    }
    return tasks;
  }

  /** 等待任务完成 */
  async awaitTask(taskId: string, timeout?: number): Promise<TaskResult> {
    const startTime = Date.now();
    const pollInterval = 100;

    while (true) {
      const task = this.getTask(taskId);
      if (!task) {
        throw new Error(`Task ${taskId} not found`);
      }

      if (task.status === 'completed' || task.status === 'failed' || task.status === 'cancelled') {
        return task;
      }

      if (timeout && Date.now() - startTime > timeout) {
        throw new Error(`Task ${taskId} timed out`);
      }

      await new Promise(resolve => setTimeout(resolve, pollInterval));
    }
  }

  private generateTaskId(): string {
    return `task_${Date.now()}_${Math.random().toString(36).slice(2)}`;
  }
}

// 全局 Task 运行器
let globalTaskRunner: TaskRunner | null = null;

export function getTaskRunner(): TaskRunner {
  if (!globalTaskRunner) {
    globalTaskRunner = new TaskRunner();
  }
  return globalTaskRunner;
}
```

### Phase 7: MCP 集成增强 (Week 7-8)

#### 7.1 MCP 桥接实现

```typescript
// packages/agent-core/src/mcp/bridge.ts
import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import { StdioClientTransport } from '@modelcontextprotocol/sdk/client/stdio.js';
import type { Tool as MCPTool } from '@modelcontextprotocol/sdk/types.js';
import type { Tool, ToolResult } from '../tool.js';
import { mcpToolToStandard } from './tool-adapter.js';

export interface MCPServerConfig {
  name: string;
  command?: string;
  args?: string[];
  env?: Record<string, string>;
  url?: string;
  autoConnect?: boolean;
}

export class MCPBridge {
  private clients: Map<string, Client> = new Map();
  private transports: Map<string, any> = new Map();
  private config: MCPServerConfig[];
  private tools: Tool[] = [];
  private toolCallbacks: Set<(tools: Tool[]) => void> = new Set();

  constructor(config: MCPServerConfig[]) {
    this.config = config;
  }

  async connect(): Promise<void> {
    for (const serverConfig of this.config) {
      await this.connectServer(serverConfig);
    }
  }

  private async connectServer(serverConfig: MCPServerConfig): Promise<void> {
    const { name, command, args, env, url } = serverConfig;

    const client = new Client({
      name: 'upup-agent-core',
      version: '0.1.0',
    }, {
      capabilities: {
        tools: {},
        resources: {},
      },
    });

    let transport;
    if (command && args) {
      transport = new StdioClientTransport({
        command,
        args,
        env,
      });
    } else if (url) {
      // HTTP/SSE transport
      const { SSEClientTransport } = await import('@modelcontextprotocol/sdk/client/sse.js');
      transport = new SSEClientTransport(new URL(url));
    }

    if (!transport) {
      throw new Error(`No valid transport for MCP server ${name}`);
    }

    await client.connect(transport);
    this.clients.set(name, client);
    this.transports.set(name, transport);

    // 获取初始工具列表
    await this.updateTools(name);
  }

  private async updateTools(serverName: string): Promise<void> {
    const client = this.clients.get(serverName);
    if (!client) return;

    const response = await client.request(
      { method: 'tools/list' },
      { method: 'tools/list', params: {} }
    );

    const mcpTools: MCPTool[] = response.tools ?? [];
    const tools = mcpTools.map(mcpTool => mcpToolToStandard(mcpTool));

    // 合并工具
    const otherTools = this.tools.filter(t => !t.name.startsWith(`${serverName}:`));
    this.tools = [...otherTools, ...tools];

    // 通知变化
    for (const callback of this.toolCallbacks) {
      callback(this.tools);
    }
  }

  getTools(): Tool[] {
    return this.tools;
  }

  async callTool(name: string, args: Record<string, unknown>): Promise<ToolResult> {
    // 查找工具所属的服务器
    for (const [serverName, client] of this.clients.entries()) {
      const toolName = `${serverName}:${name}`;
      if (this.tools.some(t => t.name === toolName || t.name === name)) {
        try {
          const response = await client.request(
            { method: 'tools/call' },
            {
              method: 'tools/call',
              params: {
                name,
                arguments: args,
              },
            }
          );

          return {
            success: true,
            output: JSON.stringify(response),
          };
        } catch (error) {
          return {
            success: false,
            error: error instanceof Error ? error.message : String(error),
          };
        }
      }
    }

    return {
      success: false,
      error: `Tool ${name} not found`,
    };
  }

  onToolsChanged(callback: (tools: Tool[]) => void): () => void {
    this.toolCallbacks.add(callback);
    return () => this.toolCallbacks.delete(callback);
  }

  async disconnect(): Promise<void> {
    for (const [name, client] of this.clients.entries()) {
      await client.close();
    }
    this.clients.clear();
    this.transports.clear();
    this.tools = [];
  }
}
```
