#### P0: Auto-Trigger 机制

```typescript
// src/skills/auto-trigger.ts
export interface AutoTriggerConfig {
  enabled: boolean;
  mode: 'auto' | 'suggest' | 'manual';
  triggers: Record<string, string[]>;
}

// 意图检测
export async function detectIntent(input: string): Promise<Intent[]> {
  const intents: Intent[] = [];
  
  // 1. Ticker 检测
  const tickers = extractTickers(input);
  if (tickers.length > 0) intents.push({ type: 'ticker', value: tickers });
  
  // 2. 意图匹配
  for (const [skill, triggers] of Object.entries(config.triggers)) {
    for (const trigger of triggers) {
      if (input.includes(trigger)) {
        intents.push({ type: 'skill', value: skill, trigger });
      }
    }
  }
  
  return intents;
}

// Skills 匹配与执行
export async function matchAndExecuteSkills(intents: Intent[]): Promise<void> {
  for (const intent of intents) {
    if (intent.type === 'skill') {
      const skill = getSkill(intent.value);
      if (skill.auto || config.mode === 'auto') {
        await executeSkill(skill, intent);
      } else if (config.mode === 'suggest') {
        await suggestSkill(skill, intent);
      }
    }
  }
}
```
